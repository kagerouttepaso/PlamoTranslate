cmake_minimum_required(VERSION 3.22)
project(mockingword-desktop-native LANGUAGES C CXX)

string(TOLOWER "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}" compiler_architecture)
if(NOT WIN32 OR NOT compiler_architecture STREQUAL "x64" OR NOT MSVC)
    message(FATAL_ERROR "Windows x64 の MSVC または clang-cl が必要です")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)

find_package(JNI REQUIRED COMPONENTS JVM)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_subdirectory("${REPO_ROOT}/third_party/llama.cpp" build-llama)

add_library(plamo-llm SHARED "${REPO_ROOT}/core-llm/src/main/cpp/plamo_llm.cpp")
target_include_directories(plamo-llm PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_options(plamo-llm PRIVATE /utf-8)
target_link_libraries(plamo-llm PRIVATE llama ggml)

add_library(mockingword-audio SHARED wasapi_capture.cpp)
target_include_directories(mockingword-audio PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_options(mockingword-audio PRIVATE /utf-8)
target_link_libraries(mockingword-audio PRIVATE mmdevapi ole32)

set(pdfium_root "${REPO_ROOT}/third_party/pdfium/chromium-8165-win-x64")
if(NOT EXISTS "${pdfium_root}/PDFiumConfig.cmake")
    message(FATAL_ERROR "tools/get-pdfium.ps1 で PDFium を取得してください")
endif()
find_package(PDFium CONFIG REQUIRED PATHS "${pdfium_root}" NO_DEFAULT_PATH)
configure_file("${pdfium_root}/bin/pdfium.dll" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/pdfium.dll" COPYONLY)

set(harfbuzz_src "${REPO_ROOT}/third_party/harfbuzz/harfbuzz-12.3.2/src")
if(NOT EXISTS "${harfbuzz_src}/harfbuzz-subset.cc")
    message(FATAL_ERROR "tools/get-harfbuzz.ps1 で HarfBuzz を取得してください")
endif()
# リリース付属の統合ソースには HarfBuzz 本体とサブセットが入っている。
add_library(harfbuzz-subset STATIC "${harfbuzz_src}/harfbuzz-subset.cc")
target_include_directories(harfbuzz-subset PUBLIC "${harfbuzz_src}")
target_compile_definitions(harfbuzz-subset PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS)
target_compile_options(harfbuzz-subset PRIVATE /utf-8 /bigobj /wd4244 /wd4267)

add_library(mockingword-pdf SHARED pdfium_jni.cpp)
target_include_directories(mockingword-pdf PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_options(mockingword-pdf PRIVATE /utf-8)
target_link_libraries(mockingword-pdf PRIVATE pdfium harfbuzz-subset)

add_library(mockingword-window SHARED window_frame.cpp)
target_include_directories(mockingword-window PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_options(mockingword-window PRIVATE /utf-8)
target_link_libraries(mockingword-window PRIVATE user32)
